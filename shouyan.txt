棋盘标点
python3 - <<'EOF'
import cv2
img = cv2.imread('/root/vision_ws/board.jpg')
g = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
ok, pts = cv2.findChessboardCorners(g, (9, 6),
        cv2.CALIB_CB_ADAPTIVE_THRESH + cv2.CALIB_CB_NORMALIZE_IMAGE)
print('detected:', ok, 0 if pts is None else len(pts))
if ok:
    pts = cv2.cornerSubPix(g, pts, (11, 11), (-1, -1),
          (cv2.TERM_CRITERIA_EPS + cv2.TERM_CRITERIA_MAX_ITER, 30, 0.001))
    for i, (u, v) in enumerate(pts.reshape(-1, 2)):
        print('corner %2d  u=%6.1f  v=%6.1f' % (i, u, v))
EOF
